calc_climate_stats <- function(data) {
  
  safe_median <- function(x) {
    x <- x[!is.na(x)]
    if (length(x) == 0) return(NA_real_)
    return(median(x))
  }
  
  # --- Rainfall stats ---
  yearly <- data$rain %>%
    group_by(Year) %>%
    summarise(
      Rainfall = sum(Rainfall.amount..millimetres., na.rm = TRUE),
      Days_Rain = sum(Rainfall.amount..millimetres. > 0, na.rm = TRUE),
      .groups = "drop"
    )
  
  rain_med <- if (nrow(yearly) == 0) NA_real_ else safe_median(yearly$Rainfall)
  days_med <- if (nrow(yearly) == 0) NA_real_ else safe_median(yearly$Days_Rain)
  
  # --- Solar ---
  solar_med <- safe_median(data$solar)
  
  # --- Temperature ---
  if (!is.null(data$temp) && "Maximum.temperature..Degree.C." %in% names(data$temp)) {
    temp_med <- safe_median(data$temp$Maximum.temperature..Degree.C.)
  } else {
    temp_med <- NA_real_
  }
  
  # --- Output (force numeric scalars) ---
  data.frame(
    median_rainfall = as.numeric(rain_med),
    median_days = as.numeric(days_med),
    median_solar = as.numeric(solar_med),
    median_temp = as.numeric(temp_med)
  )
}
  
  climate_results <- lapply(climate_data, calc_climate_stats)
  climate_results_df <- do.call(rbind, climate_results)
  climate_results_df$Location <- rownames(climate_results_df)
  rownames(climate_results_df) <- NULL
